// 双指针总结

// 1.对撞指针
核心思想：left 从最左，right 从最右，根据当前状态决定移动哪一边，直到相遇。
适用条件：数组有序，或问题具有单调性。
         两数之和、三数之和、盛水容器、验证回文串。

int left = 0, right = n - 1;
while (left < right) {
    if (满足条件) {
        // 记录答案
        left++; right--;
    } else if (需要增大) {
        left++;
    } else {
        right--;
    }
}


// 2.快慢指针
核心思想：fast 负责遍历，slow 负责维护“有效结果”。fast 一直走，满足条件就把值写到 slow 位置，slow++。
适用条件：原地删除元素、去重、移动零、压缩数组。
         链表找中点、判环、找倒数第 k 个。
// 数组
int slow = 0;
for (int fast = 0; fast < n; fast++) {
    if (满足保留条件) {
        nums[slow++] = nums[fast];
    }
}
return slow; // 新长度
// 链表
ListNode *slow = head, *fast = head;
while (fast && fast->next) {
    slow = slow->next;
    fast = fast->next->next;
}


// 3.滑动窗口
核心思想：right 负责扩张窗口，left 负责收缩窗口。窗口内维护一个状态（和、计数、哈希表等）。
经典例题：无重复字符的最长子串：窗口内用哈希表计数，出现重复就收缩 left。
         最小覆盖子串：用 need 和 window 计数，满足覆盖时收缩并更新。
         长度最小的子数组：和 >= target 时收缩并更新。
         找到字符串中所有字母异位词：固定窗口大小，滑动比较。
// 求最短
int left = 0, ans = 0;
for (int right = 0; right < n; right++) {
    // 1. 加入 nums[right]
    while (窗口不满足条件) {
        // 2. 移除 nums[left]
        left++;
    }
    // 3. 更新最长答案
    ans = max(ans, right - left + 1);
}
// 求最长
int left = 0, ans = INF;
for (int right = 0; right < n; right++) {
    // 1. 加入 nums[right]
    while (窗口满足条件) {
        // 2. 更新最短答案
        ans = min(ans, right - left + 1);
        // 3. 移除 nums[left]
        left++;
    }
}


// 4.分离双指针
核心思想：两个指针分别指向两个序列，谁小移动谁，或者按条件移动。

int i = 0, j = 0;
while (i < n && j < m) {
    if (a[i] < b[j]) {
        i++;
    } else if (a[i] > b[j]) {
        j++;
    } else {
        // 相等处理
        i++; j++;
    }
}
